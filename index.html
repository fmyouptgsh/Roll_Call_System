async function checkIn(button, studentId, studentName, ssId) { if (button.classList.contains('checked')) { return; } const originalText = button.textContent; button.textContent = '點名中...'; button.disabled = true; try { const url = getApiUrl(ssId); const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, // ← 必須加 body: JSON.stringify({ id: studentId, name: studentName }) }); const result = await response.json(); if (result.status !== "success") { throw new Error(result.message); } button.textContent = `${studentName} (已點名)`; button.classList.add('checked'); sessionStorage.setItem(`checked_${ssId}_${studentId}`, 'true'); } catch (error) { alert('點名失敗：' + error.message); button.textContent = originalText; button.disabled = false; console.error('Check-in error:', error); } }
